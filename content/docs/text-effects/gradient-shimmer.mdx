# Gradient Shimmer

Source component: `text-effects/gradient-shimmer.html`

```html
<!--
    Gradient Shimmer

    Customize the component by editing the values in x-data:
    gradient: sunrise | bubble | peach | tonic | mint | spring | twilight | bay
    easing: smooth | gentle | snappy
-->
<span
    x-data="{
        gradient: 'sunrise',
        easing: 'smooth',
        duration: 1.45,
        spread: 3,
        angle: 105,
        pauseBetween: 1000,
        baseColor: 'currentColor',
        pauseOnScroll: true,
        pauseWhenOffscreen: true,
        respectReducedMotion: true,
        presets: {
            sunrise: [
                { color: '#B6D3EF', position: 0 },
                { color: '#CAD1D7', position: 0.153 },
                { color: '#D7CFC8', position: 0.252 },
                { color: '#E1CDB9', position: 0.341 },
                { color: '#EAC6A5', position: 0.424 },
                { color: '#EDB185', position: 0.505 },
                { color: '#EF9B62', position: 0.586 },
                { color: '#F18F60', position: 0.669 },
                { color: '#F48D7A', position: 0.758 },
                { color: '#F78A94', position: 0.857 },
                { color: '#F888A0', position: 1 },
            ],
            bubble: [
                { color: '#F5EBD9', position: 0 },
                { color: '#F2D4DB', position: 0.31 },
                { color: '#EBBDDE', position: 0.5 },
                { color: '#CCBAE3', position: 0.65 },
                { color: '#8CBFF0', position: 0.82 },
                { color: '#78B0FF', position: 1 },
            ],
            peach: [
                { color: '#D9F5FA', position: 0 },
                { color: '#FCD9D6', position: 0.31 },
                { color: '#FCBAC9', position: 0.61 },
                { color: '#F0B3F5', position: 1 },
            ],
            tonic: [
                { color: '#E3EDF0', position: 0 },
                { color: '#E8EBB8', position: 0.27 },
                { color: '#F0DEA3', position: 0.43 },
                { color: '#E8B078', position: 0.75 },
                { color: '#F29682', position: 1 },
            ],
            mint: [
                { color: '#DECEE8', position: 0 },
                { color: '#CBBAEE', position: 0.21 },
                { color: '#7DC0FB', position: 0.46 },
                { color: '#00C7A6', position: 1 },
            ],
            spring: [
                { color: '#F7D5C5', position: 0.07 },
                { color: '#46A8C0', position: 0.58 },
                { color: '#43AE7D', position: 1 },
            ],
            twilight: [
                { color: '#E3CCE6', position: 0 },
                { color: '#4E8CD5', position: 0.35 },
                { color: '#6068C2', position: 0.64 },
                { color: '#38364E', position: 1 },
            ],
            bay: [
                { color: '#DBE3D0', position: 0 },
                { color: '#8DB8A7', position: 0.23 },
                { color: '#2D8E9A', position: 0.42 },
                { color: '#076492', position: 0.59 },
                { color: '#154288', position: 0.79 },
                { color: '#262C81', position: 1 },
            ],
        },
        easingPresets: {
            smooth: 'cubic-bezier(0.45, 0, 0.55, 1)',
            gentle: 'cubic-bezier(0.76, 0, 0.24, 1)',
            snappy: 'cubic-bezier(0.3, 0, 0.2, 1)',
        },
        ready: false,
        staticGradient: false,
        supported: true,
        backgroundImage: '',
        backgroundSize: '100% 100%',
        backgroundPosition: 'center',
        spreadPx: 0,
        spreadMidPx: 0,
        start: 0,
        end: 0,
        layerWidth: 1,
        animation: null,
        pauseTimer: null,
        stopVisibility: null,
        active: true,
        cancelled: false,
        bandCoreRatio: 0.44,
        spreadMidRatio: 0.72,
        maxSpreadPx: 48,

        finiteOr(value, fallback) {
            return Number.isFinite(value) ? value : fallback
        },

        resolveStops() {
            return this.presets[this.gradient] || this.presets.sunrise
        },

        buildBandGradient() {
            const stops = [...this.resolveStops()].sort((a, b) => a.position - b.position)
            const first = stops[0]?.color || 'white'
            const last = stops[stops.length - 1]?.color || 'white'
            const core = stops.map((stop) => {
                const factor = (stop.position - 0.5) * 2 * this.bandCoreRatio
                return `${stop.color} calc(50% + var(--gs-spread-mid) * ${factor.toFixed(4)})`
            }).join(', ')

            return [
                `linear-gradient(${this.angle}deg`,
                'var(--gs-base) calc(50% - var(--gs-spread))',
                `color-mix(in oklab, var(--gs-base) 42%, ${first}) calc(50% - var(--gs-spread-mid))`,
                core,
                `color-mix(in oklab, var(--gs-base) 42%, ${last}) calc(50% + var(--gs-spread-mid))`,
                'var(--gs-base) calc(50% + var(--gs-spread)))',
            ].join(', ')
        },

        supportsBackgroundClipText() {
            if (typeof window === 'undefined') return true
            if (typeof window.CSS?.supports !== 'function') return false
            return window.CSS.supports('background-clip', 'text') || window.CSS.supports('-webkit-background-clip', 'text')
        },

        prefersReducedMotion() {
            return typeof window !== 'undefined' && typeof window.matchMedia === 'function' && window.matchMedia('(prefers-reduced-motion: reduce)').matches
        },

        styleText() {
            const base = [
                'position: relative',
                'display: inline-block',
                `--gs-base: ${this.baseColor}`,
                'background-color: var(--gs-base)',
                `--gs-spread: ${this.spreadPx}px`,
                `--gs-spread-mid: ${this.spreadMidPx}px`,
            ]

            if (!this.ready || !this.supported) return base.join('; ')

            return [
                ...base,
                `background-image: ${this.backgroundImage}`,
                'background-repeat: no-repeat',
                `background-size: ${this.backgroundSize}`,
                `background-position: ${this.backgroundPosition}`,
                'background-clip: text',
                '-webkit-background-clip: text',
                '-webkit-text-fill-color: transparent',
            ].join('; ')
        },

        measure() {
            const el = this.$refs.shimmer
            const textWidth = el.getBoundingClientRect().width || 96
            const fontSize = Number.parseFloat(getComputedStyle(el).fontSize) || 14
            const fontScale = fontSize / 14
            const textLength = el.textContent?.trim().length || 1
            const spreadPx = Math.min(textLength * this.spread * fontScale, this.maxSpreadPx * fontScale)
            const layerWidth = Math.max(1, textWidth + spreadPx * 2)

            this.spreadPx = spreadPx
            this.spreadMidPx = spreadPx * this.spreadMidRatio
            this.layerWidth = layerWidth
            this.start = -spreadPx - layerWidth / 2
            this.end = textWidth + spreadPx - layerWidth / 2
            this.backgroundSize = this.staticGradient ? '100% 100%' : `${layerWidth}px 100%`
            this.backgroundPosition = this.staticGradient ? 'center' : `${this.start}px center`
        },

        observeActive(onChange) {
            if (typeof window === 'undefined') return () => {}

            let inViewport = !this.pauseWhenOffscreen || typeof IntersectionObserver === 'undefined'
            let pageVisible = !document.hidden
            let notScrolling = true
            const compute = () => onChange(inViewport && pageVisible && notScrolling)
            let observer

            if (this.pauseWhenOffscreen && typeof IntersectionObserver !== 'undefined') {
                observer = new IntersectionObserver((entries) => {
                    const entry = entries[entries.length - 1]
                    if (!entry) return
                    inViewport = entry.isIntersecting
                    compute()
                }, { rootMargin: '160px' })
                observer.observe(this.$refs.shimmer)
            }

            const onVisibility = () => {
                pageVisible = !document.hidden
                compute()
            }
            document.addEventListener('visibilitychange', onVisibility)

            let scrollTimer
            const onScroll = () => {
                notScrolling = false
                compute()
                clearTimeout(scrollTimer)
                scrollTimer = setTimeout(() => {
                    notScrolling = true
                    compute()
                }, 120)
            }
            const scrollOptions = { passive: true, capture: true }
            if (this.pauseOnScroll) window.addEventListener('scroll', onScroll, scrollOptions)

            compute()

            return () => {
                observer?.disconnect()
                document.removeEventListener('visibilitychange', onVisibility)
                if (this.pauseOnScroll) window.removeEventListener('scroll', onScroll, { capture: true })
                clearTimeout(scrollTimer)
            }
        },

        runSweep() {
            if (this.cancelled || this.staticGradient) return
            this.measure()
            const next = this.$refs.shimmer.animate([
                { backgroundPosition: `${this.start}px center` },
                { backgroundPosition: `${this.end}px center` },
            ], {
                duration: this.finiteOr(this.duration, 1.45) * 1000,
                easing: this.easingPresets[this.easing] || this.easingPresets.smooth,
                fill: 'forwards',
            })
            if (!this.active) next.pause()
            this.animation?.cancel()
            this.animation = next
            next.onfinish = () => {
                this.pauseTimer = setTimeout(() => this.runSweep(), Math.max(0, this.pauseBetween))
            }
        },

        init() {
            this.supported = this.supportsBackgroundClipText()
            this.backgroundImage = this.buildBandGradient()
            this.staticGradient = !this.supported || (this.respectReducedMotion && this.prefersReducedMotion()) || typeof this.$refs.shimmer.animate !== 'function'
            this.measure()
            this.ready = true

            if (!this.supported) return
            this.stopVisibility = this.observeActive((active) => {
                this.active = active
                if (this.animation) active ? this.animation.play() : this.animation.pause()
            })
            this.runSweep()
        },

        destroy() {
            this.cancelled = true
            this.animation?.cancel()
            clearTimeout(this.pauseTimer)
            this.stopVisibility?.()
        },
    }"
    x-ref="shimmer"
    x-init="init()"
    x-bind:style="styleText()"
    x-on:alpine:destroy="destroy()"
    class="text-on-surface-strong dark:text-on-surface-dark-strong"
>
    Gradient Shimmer
</span>

```
